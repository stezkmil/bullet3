cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BulletGpuXpbdTgs LANGUAGES CXX)
    option(INSTALL_LIBS "Install the GPU XPBD TGS runtime" ON)
endif()
# This runtime does not use Bullet's directory-wide TBB/allocator libraries.
set_property(DIRECTORY PROPERTY LINK_LIBRARIES "")
if(NOT WIN32)
    message(FATAL_ERROR "GPU XPBD TGS currently requires Windows.")
endif()
if(XPBD_PTX_COMPILER)
    find_package(CUDAToolkit 11.7 CONFIG REQUIRED)
    if(NOT EXISTS "${XPBD_PTX_COMPILER}")
        message(FATAL_ERROR "XPBD_PTX_COMPILER does not exist: ${XPBD_PTX_COMPILER}")
    endif()
    set(xpbd_ptx_command "${XPBD_PTX_COMPILER}")
    set(xpbd_ptx_dependency "${XPBD_PTX_COMPILER}")
else()
    find_package(CUDAToolkit 11.7 MODULE REQUIRED)
    if(CMAKE_CROSSCOMPILING OR NOT TARGET CUDA::nvrtc)
        message(FATAL_ERROR "Supply XPBD_PTX_COMPILER or use a native CUDA toolkit with NVRTC.")
    endif()
    add_executable(XpbdCompilePtx compile_ptx.cpp)
    target_link_libraries(XpbdCompilePtx PRIVATE CUDA::nvrtc)
    target_compile_features(XpbdCompilePtx PRIVATE cxx_std_17)
    set_target_properties(XpbdCompilePtx PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    file(GLOB nvrtc_dlls "${CUDAToolkit_BIN_DIR}/nvrtc*.dll" "${CUDAToolkit_BIN_DIR}/x64/nvrtc*.dll")
    list(LENGTH nvrtc_dlls nvrtc_dll_count)
    if(nvrtc_dll_count LESS 2)
        message(FATAL_ERROR "NVRTC and its matching builtins DLL must be installed.")
    endif()
    add_custom_command(TARGET XpbdCompilePtx POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${nvrtc_dlls} "$<TARGET_FILE_DIR:XpbdCompilePtx>" VERBATIM)
    set(xpbd_ptx_command "$<TARGET_FILE:XpbdCompilePtx>")
    set(xpbd_ptx_dependency XpbdCompilePtx)
endif()
set(ptx_header "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/xpbd_ptx.h")
add_custom_command(OUTPUT "${ptx_header}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>"
    COMMAND "${xpbd_ptx_command}" "${CMAKE_CURRENT_SOURCE_DIR}/kernels.cu" "${ptx_header}"
    DEPENDS ${xpbd_ptx_dependency} kernels.cu physx_gjk.cuh VERBATIM)
add_library(BulletGpuXpbdTgs SHARED backend.cpp api_adapter.cpp "${ptx_header}")
target_include_directories(BulletGpuXpbdTgs PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>" "../../src/BulletSoftBody")
target_link_libraries(BulletGpuXpbdTgs PRIVATE CUDA::cuda_driver)
target_compile_features(BulletGpuXpbdTgs PRIVATE cxx_std_17)
# Mesh validation and cooking must retain finite-value checks in the parent build.
target_compile_options(BulletGpuXpbdTgs PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/fp:precise>")
set_target_properties(BulletGpuXpbdTgs PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
    DEBUG_POSTFIX "" RELEASE_POSTFIX "" RELWITHDEBINFO_POSTFIX "" MINSIZEREL_POSTFIX "")
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set_target_properties(BulletGpuXpbdTgs PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
endif()
if(INSTALL_LIBS)
    install(TARGETS BulletGpuXpbdTgs RUNTIME DESTINATION bin COMPONENT GpuXpbdTgs)
    install(FILES "../../src/BulletSoftBody/btGpuXpbdTgsApi.h" DESTINATION include/bullet/BulletSoftBody COMPONENT GpuXpbdTgs)
    install(FILES "../../src/BulletSoftBody/btGpuXpbdTgsBackend.h" DESTINATION include/bullet/BulletSoftBody COMPONENT GpuXpbdTgs)
    install(FILES LICENSE-PhysX.txt LICENSE-PositionBasedDynamics.txt DESTINATION licenses/gpu-xpbd-tgs COMPONENT GpuXpbdTgs)
endif()
add_executable(XpbdCoreTest core_test.cpp)
target_include_directories(XpbdCoreTest PRIVATE "../../src/BulletSoftBody")
target_link_libraries(XpbdCoreTest PRIVATE BulletGpuXpbdTgs)
set_target_properties(XpbdCoreTest PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_test(NAME XpbdCoreTest COMMAND XpbdCoreTest)
add_executable(XpbdLoaderTest core_test.cpp)
target_compile_definitions(XpbdLoaderTest PRIVATE XPBD_TEST_RUNTIME_LOADER=1)
target_include_directories(XpbdLoaderTest PRIVATE "../../src/BulletSoftBody")
target_compile_features(XpbdLoaderTest PRIVATE cxx_std_17)
set_target_properties(XpbdLoaderTest PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_dependencies(XpbdLoaderTest BulletGpuXpbdTgs)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set_target_properties(XpbdCoreTest XpbdLoaderTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
endif()
add_test(NAME XpbdLoaderTest COMMAND XpbdLoaderTest)


# The public ABI remains double; test expected roundoff and overflow against GPU storage.
target_compile_definitions(XpbdCoreTest PRIVATE XPBD_TEST_FP32=1)
target_compile_definitions(XpbdLoaderTest PRIVATE XPBD_TEST_FP32=1)
